  <script>

    let counters = [];

    let currentCarouselId = null;

    let currentSlideIndex = 0;

    let currentGalleryId = null;


    // データ初期化とV1からのマイグレーション

    function initData() {

      const raw = localStorage.getItem('visual_counters');

      if (raw) {

        let parsed = JSON.parse(raw);

        counters = parsed.map(c => {

          if (!c.images) c.images = [];

          // 古いバージョンの 'image' プロパティを新しい配列形式へ変換

          if (c.image) {

            c.images.push({ id: generateId(), url: c.image, isCover: true });

            delete c.image;

          }

          if (c.target === undefined) c.target = 0; // 目標回数初期値

          return c;

        });

      }

      saveData();

      render();

    }


    function saveData() { localStorage.setItem('visual_counters', JSON.stringify(counters)); }

    function triggerHaptic() { if (navigator.vibrate) navigator.vibrate(40); }

    function generateId() { return '_' + Math.random().toString(36).substr(2, 9); }


    // --- カウンター基本操作 ---

    function addCounter() {

      counters.push({ id: generateId(), title: '', count: 0, target: 0, images: [] });

      saveData(); render();

    }

    function deleteCounter(id) {

      if(confirm('このボックスを完全に削除しますか？')) {

        counters = counters.filter(c => c.id !== id);

        saveData(); render();

      }

    }

    function increment(id) {

      triggerHaptic();

      const c = counters.find(x => x.id === id);

      if (c) { c.count++; saveData(); render(); }

    }

    function decrement(id) {

      triggerHaptic();

      const c = counters.find(x => x.id === id);

      if (c) { c.count--; saveData(); render(); }

    }

    function resetCount(id) {

      const c = counters.find(x => x.id === id);

      if (c && confirm('カウントをリセットしますか？')) { c.count = 0; saveData(); render(); }

    }

    function updateTitle(id, val) { const c = counters.find(x => x.id === id); if(c) { c.title = val; saveData(); } }

    function updateTarget(id, val) { const c = counters.find(x => x.id === id); if(c) { c.target = parseInt(val) || 0; saveData(); render(); } }


    // --- バックアップ（エクスポート/インポート） ---

    function exportData() {

      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(counters));

      const downloadAnchorNode = document.createElement('a');

      downloadAnchorNode.setAttribute("href", dataStr);

      downloadAnchorNode.setAttribute("download", "visual_counter_backup.json");

      document.body.appendChild(downloadAnchorNode);

      downloadAnchorNode.click();

      downloadAnchorNode.remove();

    }

    function importData(event) {

      const file = event.target.files[0];

      if (!file) return;

      const reader = new FileReader();

      reader.onload = function(e) {

        try {

          counters = JSON.parse(e.target.result);

          saveData(); render(); alert('復元が完了しました。');

        } catch (err) { alert('ファイルの読み込みに失敗しました。'); }

      };

      reader.readAsText(file);

      event.target.value = '';

    }


    // --- 画像圧縮・アップロード ---

    function processImage(file, callback) {

      const reader = new FileReader();

      reader.onload = function(e) {

        const img = new Image();

        img.onload = function() {

          const canvas = document.createElement('canvas');

          const MAX_SIZE = 500; // 複数枚保存のため圧縮率を強化

          let w = img.width, h = img.height;

          if (w > h) { if (w > MAX_SIZE) { h *= MAX_SIZE / w; w = MAX_SIZE; } } 

          else { if (h > MAX_SIZE) { w *= MAX_SIZE / h; h = MAX_SIZE; } }

          canvas.width = w; canvas.height = h;

          const ctx = canvas.getContext('2d');

          ctx.drawImage(img, 0, 0, w, h);

          callback(canvas.toDataURL('image/jpeg', 0.6));

        };

        img.src = e.target.result;

      };

      reader.readAsDataURL(file);

    }


    function uploadToGallery(event) {

      const file = event.target.files[0];

      if (!file || !currentGalleryId) return;

      processImage(file, (dataUrl) => {

        const c = counters.find(x => x.id === currentGalleryId);

        const isFirst = c.images.length === 0;

        c.images.push({ id: generateId(), url: dataUrl, isCover: isFirst });

        saveData();

        renderGallery();

        render(); // 背景更新

      });

      event.target.value = '';

    }


    // --- ギャラリー管理 ---

    function openGallery(id) { currentGalleryId = id; renderGallery(); document.getElementById('galleryModal').style.display = 'flex'; }

    function closeGallery() { document.getElementById('galleryModal').style.display = 'none'; currentGalleryId = null; }

    

    function setCover(imageId) {

      const c = counters.find(x => x.id === currentGalleryId);

      c.images.forEach(img => img.isCover = (img.id === imageId));

      saveData(); renderGallery(); render();

    }

    function deleteImage(imageId) {

      if(!confirm('この画像を削除しますか？')) return;

      const c = counters.find(x => x.id === currentGalleryId);

      c.images = c.images.filter(img => img.id !== imageId);

      if(c.images.length > 0 && !c.images.some(img => img.isCover)) {

        c.images[0].isCover = true; // カバーが消えたら最初の画像をカバーに

      }

      saveData(); renderGallery(); render();

    }


    function renderGallery() {

      const c = counters.find(x => x.id === currentGalleryId);

      const grid = document.getElementById('galleryGrid');

      grid.innerHTML = '';

      c.images.forEach(img => {

        const div = document.createElement('div');

        div.className = `gallery-item ${img.isCover ? 'is-cover' : ''}`;

        div.style.backgroundImage = `url(${img.url})`;

        div.innerHTML = `

          <div class="gallery-item-actions">

            <button class="g-btn" onclick="setCover('${img.id}')">📍</button>

            <button class="g-btn" style="color:red;" onclick="deleteImage('${img.id}')">🗑️</button>

          </div>

        `;

        grid.appendChild(div);

      });

    }


    // --- カルーセル表示 ---

    function openCarousel(id, event) {

      event.stopPropagation();

      const c = counters.find(x => x.id === id);

      if (!c || c.images.length === 0) return;

      currentCarouselId = id;

      // 表紙画像を初期スライドに設定

      const coverIndex = c.images.findIndex(img => img.isCover);

      currentSlideIndex = coverIndex >= 0 ? coverIndex : 0;

      updateCarouselView();

      document.getElementById('carouselModal').style.display = 'flex';

    }

    function closeCarousel() { document.getElementById('carouselModal').style.display = 'none'; currentCarouselId = null; }

    function changeSlide(direction) {

      const c = counters.find(x => x.id === currentCarouselId);

      currentSlideIndex += direction;

      if (currentSlideIndex < 0) currentSlideIndex = c.images.length - 1;

      if (currentSlideIndex >= c.images.length) currentSlideIndex = 0;

      updateCarouselView();

    }

    function updateCarouselView() {

      const c = counters.find(x => x.id === currentCarouselId);

      document.getElementById('modalImage').src = c.images[currentSlideIndex].url;

      document.getElementById('imageIndexDisplay').innerText = `${currentSlideIndex + 1} / ${c.images.length}`;

    }


    // --- メイン描画 ---

    function render() {

      const grid = document.getElementById('grid');

      grid.innerHTML = '';

      counters.forEach(c => {

        const wrapper = document.createElement('div');

        wrapper.className = 'counter-wrapper';

        

        let coverImg = c.images.find(img => img.isCover) || c.images[0];

        const bgStyle = coverImg ? `background-image: url(${coverImg.url});` : '';

        const isTargetReached = c.target > 0 && c.count >= c.target;

        const boxClass = `counter-box ${isTargetReached ? 'target-reached' : ''}`;


        wrapper.innerHTML = `

          <div class="${boxClass}" style="${bgStyle}" onclick="increment('${c.id}')">

            <div class="ui-layer">

              <div class="top-bar">

                <input type="text" class="title-input" value="${c.title}" placeholder="タイトル" 

                       onclick="event.stopPropagation()" onchange="updateTitle('${c.id}', this.value)">

                <input type="number" class="target-input" value="${c.target || ''}" placeholder="目標" title="目標回数"

                       onclick="event.stopPropagation()" onchange="updateTarget('${c.id}', this.value)">

              </div>

              <p class="count-display">${c.count}</p>

              <div style="position:absolute; bottom:10px; right:10px; z-index:10;">

                <button class="icon-btn" style="${c.images.length > 0 ? 'background: rgba(0,0,0,0.6)' : 'opacity:0.3'};" 

                        onclick="openCarousel('${c.id}', event)">👁️</button>

              </div>

            </div>

          </div>

          <div class="action-bar">

            <button class="icon-btn" onclick="decrement('${c.id}')">➖</button>

            <button class="icon-btn" onclick="resetCount('${c.id}')">🔄</button>

            <button class="icon-btn" onclick="openGallery('${c.id}')">🗂️</button>

            <button class="icon-btn" style="background:rgba(255,0,0,0.3)" onclick="deleteCounter('${c.id}')">🗑️</button>

          </div>

        `;

        grid.appendChild(wrapper);

      });

    }


    initData();

  </script>
